<script>
const USER="fam3000";
const REPO="mb";
const BASE=`https://${USER}.github.io/${REPO}/`;

let files=[];
let favorites=JSON.parse(localStorage.getItem("favorites")||"[]");
let recent=JSON.parse(localStorage.getItem("recent")||"[]");

// 遞迴載入倉庫所有檔案
async function loadFiles() {
    try {
        const res = await fetch(`https://api.github.com/repos/${USER}/${REPO}/git/trees/main?recursive=1`);
        if (!res.ok) throw new Error("API請求失敗");
        const data = await res.json();
        files = data.tree.filter(item => item.type !== 'tree').map(item => ({
            name: item.path.split('/').pop(),
            path: item.path,
            size: item.size,
            download_url: `https://raw.githubusercontent.com/${USER}/${REPO}/main/${item.path}`
        }));
        render(files);
        updateInfo();
    } catch (err) {
        document.getElementById("list").innerHTML = `<div class="loading">讀取失敗：${err.message}</div>`;
    }
}

// 取得檔案圖示
function getTypeIcon(filename) {
    const ext = filename.split('.').pop().toLowerCase();
    const map = {
        html: '🌐', md: '📝', mp3: '🎵', png: '🖼️', jpg: '🖼️', jpeg: '🖼️',
        pdf: '📕', zip: '📦', rar: '📦', js: '💻', css: '🎨'
    };
    return map[ext] || '📄';
}

// 渲染卡片
function render(arr){
    const list=document.getElementById("list");
    list.innerHTML="";

    if(arr.length===0){
        list.innerHTML="<div class='loading'>沒有找到檔案</div>";
        return;
    }

    arr.forEach(file=>{
        const card=document.createElement("div");
        card.className="card";
        card.onclick=(e)=>{
            if(!e.target.classList.contains("favorite")) openFile(file.path);
        };
        card.innerHTML=`
            <span class="favorite ${favorites.includes(file.path)?'active':''}" onclick="toggleFavorite(event,'${file.path}')">
                ${favorites.includes(file.path)?'⭐':'☆'}
            </span>
            <div class="title">${getTypeIcon(file.name)} ${file.name}</div>
            <div class="path">${file.path}</div>
            <div class="size">${file.size??0} bytes</div>
        `;
        list.appendChild(card);
    });
}

// 開啟檔案
async function openFile(path){
    const file=files.find(f=>f.path===path);
    if(!file)return;

    // 記錄最近開啟
    recent=recent.filter(p=>p!==path);
    recent.unshift(path);
    localStorage.setItem("recent",JSON.stringify(recent.slice(0,20)));

    const ext=path.split('.').pop().toLowerCase();
    if(ext==="md"){
        await showMarkdown(file);
    }else if(["png","jpg","jpeg","gif"].includes(ext)){
        showImage(file);
    }else if(ext==="mp3"){
        showAudio(file);
    }else{
        window.open(BASE+path,"_blank");
    }
}

// 圖片預覽
function showImage(file){
    document.getElementById("markdownContent").innerHTML=`
        <h2>${file.name}</h2><hr style="margin:12px 0">
        <img class="preview" src="${file.download_url}">
    `;
    document.getElementById("markdown").style.display="block";
}

// 音樂播放
function showAudio(file){
    document.getElementById("markdownContent").innerHTML=`
        <h2>${file.name}</h2><hr style="margin:12px 0">
        <audio controls src="${file.download_url}">
    `;
    document.getElementById("markdown").style.display="block";
}

// Markdown解析
function markdownParser(text){
    return text
    .replace(/^# (.*$)/gim,"<h1>$1</h1>")
    .replace(/^## (.*$)/gim,"<h2>$1</h2>")
    .replace(/\*\*(.*?)\*\*/g,"<b>$1</b>")
    .replace(/`(.*?)`/g,"<code>$1</code>")
    .replace(/\n/g,"<br>");
}

async function showMarkdown(file){
    const res=await fetch(file.download_url);
    const text=await res.text();
    document.getElementById("markdownContent").innerHTML=`
        <h2>${file.name}</h2><hr style="margin:12px 0">
        <div>${markdownParser(text)}</div>
    `;
    document.getElementById("markdown").style.display="block";
}

function closeMarkdown(){
    document.getElementById("markdown").style.display="none";
}
document.getElementById("markdown").onclick=e=>{
    if(e.target.id==="markdown") closeMarkdown();
};

// 搜尋
document.getElementById("search").oninput=function(){
    const k=this.value.toLowerCase().trim();
    if(!k){render(files);return;}
    const res=files.filter(f=>f.name.toLowerCase().includes(k)||f.path.toLowerCase().includes(k));
    render(res);
};

// 收藏
function toggleFavorite(event,path){
    event.stopPropagation();
    if(favorites.includes(path)){
        favorites=favorites.filter(x=>x!==path);
    }else{
        favorites.push(path);
    }
    localStorage.setItem("favorites",JSON.stringify(favorites));
    render(files);
}

function showFavorites(){
    render(files.filter(f=>favorites.includes(f.path)));
}

function showRecent(){
    let result=[];
    recent.forEach(p=>{
        const item=files.find(f=>f.path===p);
        if(item)result.push(item);
    });
    render(result);
}

// 排序
function sortFiles(type){
    let list=[...files];
    if(type==="name") list.sort((a,b)=>a.name.localeCompare(b.name));
    if(type==="size") list.sort((a,b)=>(b.size||0)-(a.size||0));
    render(list);
}

// 更新資訊文字
function updateInfo(){
    document.getElementById("info").innerHTML=`全部檔案：${files.length} &nbsp;&nbsp;收藏：${favorites.length}`;
}

// 啟動
loadFiles();
